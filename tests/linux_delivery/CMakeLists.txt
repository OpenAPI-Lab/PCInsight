cmake_minimum_required(VERSION 3.20)
project(pcinsight_linux_delivery LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This delivery audit requires a Linux target")
endif()
if(NOT IS_DIRECTORY "${PCINSIGHT_AUDIT_PREFIX}")
  message(FATAL_ERROR "Set PCINSIGHT_AUDIT_PREFIX to a dedicated installed SDK prefix")
endif()
file(REAL_PATH "${PCINSIGHT_AUDIT_PREFIX}" audit_prefix)
# Refuse a workspace or system prefix before walking its tree.
file(GLOB top_entries RELATIVE "${audit_prefix}" "${audit_prefix}/*")
foreach(entry IN LISTS top_entries)
  if(NOT entry MATCHES "^(include|lib|lib64|share)$" OR NOT IS_DIRECTORY "${audit_prefix}/${entry}")
    message(FATAL_ERROR "Expected a dedicated SDK prefix; unexpected top-level entry: ${entry}")
  endif()
endforeach()
find_package(pcinsight 0.2.1 EXACT CONFIG QUIET PATHS "${audit_prefix}" NO_DEFAULT_PATH)
if(pcinsight_FOUND)
  message(FATAL_ERROR "The 0.2.0 SDK unexpectedly accepted an exact 0.2.1 request")
endif()
find_package(pcinsight 0.2.0 EXACT CONFIG REQUIRED PATHS "${audit_prefix}" NO_DEFAULT_PATH)

function(require_inside path)
  if(NOT EXISTS "${path}")
    message(FATAL_ERROR "Missing installed SDK path: ${path}")
  endif()
  file(REAL_PATH "${path}" resolved)
  cmake_path(IS_PREFIX audit_prefix "${resolved}" NORMALIZE inside)
  if(NOT inside)
    message(FATAL_ERROR "SDK path escapes install prefix: ${resolved}")
  endif()
endfunction()
require_inside("${pcinsight_CONFIG}")
get_target_property(archive pcinsight::pcinsight IMPORTED_LOCATION_RELEASE)
require_inside("${archive}")
get_target_property(headers pcinsight::pcinsight INTERFACE_INCLUDE_DIRECTORIES)
foreach(directory IN LISTS headers)
  require_inside("${directory}")
endforeach()

# Only library, headers, license and CMake metadata; no executables or units.
set(installed_license "${audit_prefix}/share/licenses/pcinsight/LICENSE")
require_inside("${installed_license}")
file(SHA256 "${installed_license}" installed_license_hash)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../../LICENSE" source_license_hash)
if(NOT installed_license_hash STREQUAL source_license_hash)
  message(FATAL_ERROR "The installed SDK license differs from the PCInsight license")
endif()
file(GLOB_RECURSE payload RELATIVE "${audit_prefix}" "${audit_prefix}/*")
foreach(path IN LISTS payload)
  require_inside("${audit_prefix}/${path}")
  if(NOT path MATCHES "^include/pcinsight/[^/]+\\.hpp$|(^|/)libpcinsight\\.a$|(^|/)cmake/pcinsight/pcinsight[^/]*\\.cmake$|^share/licenses/pcinsight/LICENSE$")
    message(FATAL_ERROR "Unexpected SDK payload: ${path}")
  endif()
endforeach()
list(LENGTH payload payload_count)
message(STATUS "Audited ${payload_count} installed files at ${audit_prefix}")

enable_testing()
add_executable(delivery-network-addresses ../linux_abi/network_address_consumer.cpp)
target_link_libraries(delivery-network-addresses PRIVATE pcinsight::pcinsight)
add_test(NAME network-addresses-sdk COMMAND delivery-network-addresses)
set_tests_properties(network-addresses-sdk PROPERTIES TIMEOUT 15 SKIP_RETURN_CODE 77)
foreach(panel IN ITEMS performance processes app_history startup users services)
  add_executable(delivery-${panel} ../../examples/linux_${panel}.cpp)
  target_link_libraries(delivery-${panel} PRIVATE pcinsight::pcinsight)
  add_test(NAME ${panel}-fixture COMMAND delivery-${panel} --self-test)
  add_test(NAME ${panel}-sample COMMAND delivery-${panel} --sample)
  set_tests_properties(${panel}-fixture ${panel}-sample PROPERTIES TIMEOUT 30)
endforeach()
