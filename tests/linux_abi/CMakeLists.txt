cmake_minimum_required(VERSION 3.20)
project(pcinsight_linux_abi LANGUAGES CXX)
find_package(pcinsight 0.2.0 EXACT CONFIG REQUIRED)
add_executable(linux-abi-probe probe.cpp)
target_link_libraries(linux-abi-probe PRIVATE pcinsight::pcinsight)
set_target_properties(linux-abi-probe PROPERTIES CXX_EXTENSIONS OFF CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  enable_testing()
  add_executable(linux-sdk-result-contract result_contract.cpp)
  target_link_libraries(linux-sdk-result-contract PRIVATE pcinsight::pcinsight)
  add_test(NAME linux-sdk-result-contract COMMAND linux-sdk-result-contract)
  set_tests_properties(linux-sdk-result-contract PROPERTIES TIMEOUT 10)
endif()
if(PCINSIGHT_ABI_BASELINE)
  file(READ "${PCINSIGHT_ABI_BASELINE}" baseline)
  string(REPLACE "\r\n" "\n" baseline "${baseline}")
  string(REGEX REPLACE "\nSystemSnapshot [0-9]+ [0-9]+\n" "\nSystemSnapshot 0 1\n" bad_layout "${baseline}")
  if(bad_layout STREQUAL baseline)
    message(FATAL_ERROR "Missing SystemSnapshot baseline row")
  endif()
  string(REGEX REPLACE "^[^\n]+" "profile deliberately-incompatible" bad_profile "${baseline}")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/bad-layout.txt" "${bad_layout}")
  string(REGEX REPLACE "\nActionResult\\.process_id offset [0-9]+\n" "\nActionResult.process_id offset 9999\n" bad_offset "${baseline}")
  if(bad_offset STREQUAL baseline)
    message(FATAL_ERROR "Missing ActionResult.process_id baseline offset")
  endif()
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/bad-offset.txt" "${bad_offset}")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/bad-profile.txt" "${bad_profile}")
  enable_testing()
  add_test(NAME linux-abi-baseline COMMAND linux-abi-probe "${PCINSIGHT_ABI_BASELINE}")
  add_test(NAME linux-abi-reject-layout COMMAND ${CMAKE_COMMAND} "-DPROBE=$<TARGET_FILE:linux-abi-probe>"
    "-DBASELINE=${CMAKE_CURRENT_BINARY_DIR}/bad-layout.txt" -DEXIT=1 "-DDIAGNOSTIC=Layout mismatch: SystemSnapshot"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/expect_rejection.cmake")
  add_test(NAME linux-abi-reject-profile COMMAND ${CMAKE_COMMAND} "-DPROBE=$<TARGET_FILE:linux-abi-probe>"
    "-DBASELINE=${CMAKE_CURRENT_BINARY_DIR}/bad-profile.txt" -DEXIT=2 "-DDIAGNOSTIC=Profile mismatch:"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/expect_rejection.cmake")
  add_test(NAME linux-abi-reject-offset COMMAND ${CMAKE_COMMAND} "-DPROBE=$<TARGET_FILE:linux-abi-probe>"
    "-DBASELINE=${CMAKE_CURRENT_BINARY_DIR}/bad-offset.txt" -DEXIT=1 "-DDIAGNOSTIC=Layout mismatch: ActionResult.process_id"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/expect_rejection.cmake")
  set_tests_properties(linux-abi-baseline linux-abi-reject-layout linux-abi-reject-profile linux-abi-reject-offset PROPERTIES TIMEOUT 10)
endif()
