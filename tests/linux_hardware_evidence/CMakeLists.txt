cmake_minimum_required(VERSION 3.20)
project(pcinsight_linux_hardware_evidence LANGUAGES CXX)
find_package(pcinsight 0.2.0 EXACT CONFIG REQUIRED)
add_executable(linux-hardware-evidence report.cpp)
target_link_libraries(linux-hardware-evidence PRIVATE pcinsight::pcinsight)
enable_testing()
add_test(NAME hardware-report-fixture COMMAND linux-hardware-evidence --self-test)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_test(NAME hardware-report-native COMMAND linux-hardware-evidence --sample)
  set_tests_properties(hardware-report-native PROPERTIES TIMEOUT 30)
endif()
set_tests_properties(hardware-report-fixture PROPERTIES TIMEOUT 10)
