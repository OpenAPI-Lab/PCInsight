# Fingerprint the public headers, including filename and bytes. Track edits as
# configure inputs so incremental builds regenerate the marker when needed.
file(GLOB_RECURSE pcinsight_public_headers CONFIGURE_DEPENDS
  RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}/include/pcinsight/*.hpp")
list(SORT pcinsight_public_headers)
set(pcinsight_header_manifest "pcinsight-sdk-headers-v1\n")
foreach(header IN LISTS pcinsight_public_headers)
  set(header_path "${CMAKE_CURRENT_SOURCE_DIR}/include/${header}")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${header_path}")
  file(SHA256 "${header_path}" header_digest)
  string(APPEND pcinsight_header_manifest "${header}=${header_digest}\n")
endforeach()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/cmake/sdk_headers.hpp.in" marker_template_digest)
string(APPEND pcinsight_header_manifest "marker-template=${marker_template_digest}\n")
string(SHA256 PCINSIGHT_HEADER_FINGERPRINT "${pcinsight_header_manifest}")
configure_file(cmake/sdk_headers.hpp.in generated/include/pcinsight/sdk_headers.hpp @ONLY)
configure_file(cmake/sdk_headers.cpp.in generated/sdk_headers.cpp @ONLY)
